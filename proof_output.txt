lemma executable:
  exists-trace
  "∃ C D rid op kh k #i #j.
    ((Exec( D, C, rid, op, kh, k ) @ #i) ∧ (HybridDevice( D ) @ #j)) ∧
    (¬(∃ A #r. RevLtk( A ) @ #r))"
/*
guarded formula characterizing all satisfying traces:
"∃ C D rid op kh k #i #j.
  (Exec( D, C, rid, op, kh, k ) @ #i) ∧ (HybridDevice( D ) @ #j)
 ∧
  ∀ A #r. (RevLtk( A ) @ #r) ⇒ ⊥"
*/

solve( !D_Session( $D, C, k ) ▶₀ #i )
  case D_2_case_1
  solve( !KU( sign(<'ODC_C', $C, $D, nc, ~nd, X, 'g'^~y, <'hybrid', o2>,
